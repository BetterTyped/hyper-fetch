  dependencyTracking?: boolean; 
  disabled?: boolean;

  refetchBlurred?: boolean;

  refresh?: boolean; 
  refreshTime?: number; 
  revalidate?: boolean; 
  select?: (context: { items: PaginationItem<R, P>[]; pages: ExtractResponseType<R>[] }) => TSelected;

  suspense?: boolean;
